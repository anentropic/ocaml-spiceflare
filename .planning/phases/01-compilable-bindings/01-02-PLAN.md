
Replace with logic that checks `d.d_extends` for single-inheritance alias candidates:

```ocaml
let type_item =
  match d.d_extends with
  | [ Ref { name = parent_name; args = [] } ]
    when Resolve.is_known_decl env parent_name
         && parent_name <> "Error"
         && not (String.equal parent_name "Record") ->

      (Ref { name = parent_name; args = [] }) in
    Type (Alias { name = "t"; target; doc = d.d_doc })

```

- `extends Error`: keeps `type t` abstract (Error maps to `Js.Exn.t`, not a bindable parent)
- `extends Record`: keeps abstract (TS utility type, not a real parent)
- Multiple extends or no extends: keeps abstract

```ocaml
  let fields = ... (* existing field extraction *) in

    | Some parent_name ->
```

**Step A: Add position-aware union classification**

Currently `classify_union` only returns `` `Nullable `` or `` `Fallback ``. Add a new function `classify_union_input` that also detects Bind candidates. Add this near the existing `classify_union`:

```ocaml
(** Classify a union in input position -- may produce Bind candidates *)
and classify_union_input members =
  let non_null =
    List.filter
      (fun m ->
         match m with
         | Ref { name = "null" | "undefined"; _ } -> false
         | _ -> true)
      members
  in
  let has_null = List.length non_null < List.length members in
  if has_null && List.length non_null = 1 then
    `Nullable (List.hd non_null)
  else if List.length non_null >= 2 && List.length non_null <= 5 then
    `Bind (non_null, has_null)
  else
    `Fallback
```

**Step D (revised): Detect Bind candidates at transform_param level**

Instead of modifying `map_type`'s return type (which would require changing the AST), detect Bind candidates when transforming parameters. Add a helper:

```ocaml
(** Check if a type is a Bind-eligible union in input position.
    Returns Some (non_null_members, has_null) if so. *)
let detect_bind_union (t : Ir.typ) =
  match t with
  | Union members ->
    let non_null =
      List.filter
        (fun m ->
           match m with
           | Ref { name = "null" | "undefined"; _ } -> false
           | _ -> true)
        members
    in
    let has_null = List.length non_null < List.length members in
    if List.length non_null >= 2 && List.length non_null <= 5 then
      Some (non_null, has_null)
    else
      None
  | _ -> None
```

Then modify `transform_member` to collect Bind modules alongside externals. When a method parameter has a Bind-eligible union, instead of mapping it to `'a Js.t`, create a `Bind` item and use `Bind.t` as the parameter type.

**Step E: Build Bind modules in transform_member**

In `transform_member`, after processing methods, scan for parameters with Bind-eligible unions. For each such parameter:

1. Build a `bind_module` with constructors derived from the union members
2. Emit the `Bind` item before the `External` that uses it
3. Replace the parameter type with a reference to the Bind module's `t`

This is complex to do inline. A pragmatic approach: add a post-processing pass in `transform_member` or `transform_class` that scans method externals for `'a Js.t` parameters that could be Bind modules.

**Simpler implementation**: Add Bind detection at the `transform_method_overload` level. For each parameter whose `p_type` is a `Union` that `detect_bind_union` matches, create a Bind module and replace the param type.

Modify `transform_method_overload` to return `external_decl * bind_module list`:

```ocaml
let transform_method_overload env ~current_module ~type_params ~ocaml_name ~js_name ~self_type
    ~source_module ~is_static (ov : method_overload) ~has_rest
    : external_decl * bind_module list =
  let bind_modules = ref [] in
  let param_index = ref 0 in
  let params =
    List.map (fun (p : Ir.param) ->
      match detect_bind_union p.p_type with
      | Some (non_null, has_null) ->
        incr param_index;
        let bm_name =
          let base = Naming.to_snake_case p.p_name in
          String.capitalize_ascii base ^ "_Bind"
        in
        let constructors =
          List.map (fun m ->
            let typ = map_type env ~current_module ~type_params m in
            let name = bind_constructor_name m in
            (name, typ))
          non_null
        in
        let bm = { bm_name; bm_constructors = constructors; bm_has_null = has_null } in
        bind_modules := bm :: !bind_modules;
        labelled_param (Naming.ocaml_name_of_member p.p_name) (Tname (bm_name ^ ".t"))
      | None ->
        transform_param env ~current_module ~type_params p)
    ov.mo_params
  in
  (* ... rest of existing logic building the external_decl ... *)
  (external_decl, List.rev !bind_modules)
```

Then in the caller (`transform_member`), emit `Bind bm` items before the `External` item for each collected bind module.
